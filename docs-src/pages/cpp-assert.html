Comprobaciones de las pruebas automáticas.

Assert

void isTrue(const T& v, std::string_view message = {})Es verdadero (bool, Value, Entity valida, puntero...).
void isFalse(const T& v, std::string_view message = {})Es falso.
void equal(const A& actual, const E& expected, std::string_view message = {})actual == expected (numeros, texto, Vec3, Value, entidades...).
void notEqual(const A& actual, const E& other, std::string_view message = {})actual != other.
void near(const A& actual, const E& expected, double tolerance = 1e-4, std::string_view message = {})|actual - expected| <= tolerance (numeros, Vec2, Vec3, Quat, Color: cada componente).
void approx(const A& actual, const E& expected, double tolerance = 1e-4, std::string_view message = {})Como near (el nombre de antes; sirve tambien con <windows.h>, que define near).
void greater(const A& a, const B& b, std::string_view message = {})a > b.
void less(const A& a, const B& b, std::string_view message = {})a < b.
void atLeast(const A& a, const B& b, std::string_view message = {})a >= b.
void atMost(const A& a, const B& b, std::string_view message = {})a <= b.
void isNull(const T& v, std::string_view message = {})Es nulo: Value nil, Entity nula o destruida, nullptr, optional vacio, asset sin poner.
void notNull(const T& v, std::string_view message = {})No es nulo.
void throws(F&& fn, std::string_view message = {})fn() lanza una excepcion.
void fail(std::string_view message = {})Falla siempre (y termina la prueba).
